   415→
   416→export default function CustomerExperienceSection() {
   417→  /* ---- state ---- */
   418→  const [channel, setChannel] = useState<ChannelType>('select');
   419→  const [step, setStep] = useState<ChatStep>('init');
   420→  const [messages, setMessages] = useState<Message[]>([]);
   421→  const [showTyping, setShowTyping] = useState(false);
   422→  const [reviewInput, setReviewInput] = useState('');
   423→  const [inputFocused, setInputFocused] = useState(false);
   424→  const [sectionVisible, setSectionVisible] = useState(false);
   425→
   426→  const chatEndRef = useRef<HTMLDivElement>(null);
   427→  const sectionRef = useRef<HTMLDivElement>(null);
   428→  const msgIdRef = useRef(0);
   429→  const inputRef = useRef<HTMLInputElement>(null);
   430→
   431→  const theme = channel !== 'select' ? getTheme(channel) : whatsappTheme;
   432→
   433→  /* ---- scroll to bottom (within chat container only) ---- */
   434→  const scrollToBottom = useCallback(() => {
   435→    requestAnimationFrame(() => {
   436→      chatEndRef.current?.scrollIntoView({ behavior: 'smooth', block: 'nearest', inline: 'nearest' });
   437→    });
   438→  }, []);
   439→
   440→  /* ---- add message helper ---- */
   441→  const addMessage = useCallback(
   442→    (sender: 'bot' | 'user', text: string) => {
   443→      const id = ++msgIdRef.current;
   444→      setMessages((prev) => [...prev, { id, sender, text, time: nowTime() }]);
   445→      return id;
   446→    },
   447→    [],
   448→  );
   449→
   450→  /* ---- auto-start on scroll into view ---- */
   451→  useEffect(() => {
   452→    const observer = new IntersectionObserver(
   453→      ([entry]) => {
   454→        if (entry.isIntersecting && !sectionVisible) {
   455→          setSectionVisible(true);
   456→        }
   457→      },
   458→      { threshold: 0.25 },
   459→    );
   460→    if (sectionRef.current) observer.observe(sectionRef.current);
   461→    return () => observer.disconnect();
   462→  }, [sectionVisible]);
   463→
   464→  /* ---- Step 1: initial bot message ---- */
   465→  useEffect(() => {
   466→    if (!sectionVisible || channel === 'select') return;
   467→    const t1 = setTimeout(() => setShowTyping(true), 300);
   468→    const t2 = setTimeout(() => {
   469→      setShowTyping(false);
   470→      addMessage(
   471→        'bot',
   472→        'Hi Sarah 👋\nThank you for choosing us!\nWe\'d love to hear how your experience was.\nWould you be willing to leave a quick rating?',
   473→      );
   474→      setStep('waiting_rating');
   475→    }, 1300);
   476→    return () => {
   477→      clearTimeout(t1);
   478→      clearTimeout(t2);
   479→    };
   480→  }, [channel, sectionVisible, addMessage]);
   481→
   482→  /* ---- scroll on new messages (only when section is visible) ---- */
   483→  useEffect(() => {
   484→    if (sectionVisible && channel !== 'select') scrollToBottom();
   485→  }, [messages, showTyping, scrollToBottom, sectionVisible, channel]);
   486→
   487→  /* ---- focus input when it appears ---- */
   488→  useEffect(() => {
   489→    if (step === 'showing_review_input') {
   490→      setTimeout(() => inputRef.current?.focus(), 400);
   491→    }
   492→  }, [step]);
